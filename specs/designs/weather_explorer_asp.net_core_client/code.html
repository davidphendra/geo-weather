app.MapPost("/api/v1/weather/notes", async (
    [FromBody] CreateWeatherNoteCommand command, 
    IMediator mediator, 
    CancellationToken ct) =>
{
    var result = await mediator.Send(command, ct);
    return Results.Created($"/api/v1/weather/notes/{result.Id}", result);
})
.WithName("CreateWeatherNote")
.WithTags("WeatherCommands")
.Produces(StatusCodes.Status201Created);

// MediatR Handler
public sealed class CreateWeatherNoteHandler(AppDbContext db, ILogger logger) 
    : IRequestHandler
{
    public async Task Handle(CreateWeatherNoteCommand cmd, CancellationToken ct)
    {
        var entity = new WeatherAnnotationEntity(cmd.CityId, cmd.Title, cmd.Notes);
        db.WeatherAnnotations.Add(entity);
        await db.SaveChangesAsync(ct);
        return WeatherNoteResponse.FromEntity(entity);
    }
}
